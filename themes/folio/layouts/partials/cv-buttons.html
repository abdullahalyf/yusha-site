{{- /* CV download buttons. With `cvs:` in profile.yaml every listed file gets its own button; otherwise the single CV from cv.html is used. Usage: partial "cv-buttons.html" (dict "profile" $p "class" "extra-class") */ -}} {{- $p := .profile -}} {{- $extra := .class | default "" -}} {{- if $p.cvs -}} {{- range $i, $c := $p.cvs -}} {{- if fileExists (printf "personal/%s" $c.file) }} {{ partial "icon.html" (dict "name" "download" "class" "btn__icon") }}{{ $c.label | default "Download CV" }} {{- end -}} {{- end -}} {{- else -}} {{- with partial "cv.html" $p }} {{ partial "icon.html" (dict "name" "download" "class" "btn__icon") }}Download CV {{- end -}} {{- end -}}
