{{- /* Favicon: SVG with the person's initials in the style accent colour. */}} {{- $p := . -}} {{- $style := site.Params.style | default "harbor" -}} {{- $accent := dict "harbor" "#0E7490" "grove" "#2F6B4F" "orchid" "#B4236B" "graphite" "#1F6FEB" "saffron" "#4338CA" "nocturne" "#0E1424" -}} {{- $color := index $accent $style | default "#0E7490" -}} {{- $words := split $p.name " " -}} {{- $initials := "" -}} {{- range $words -}} {{- if and (lt (len $initials) 2) (gt (len .) 0) -}} {{- $initials = printf "%s%s" $initials (substr . 0 1) -}} {{- end -}} {{- end -}} {{- $initials = upper $initials -}} {{- $ink := cond (eq $style "nocturne") "#F2B544" "#FFFFFF" -}} {{- $svg := printf `` $p.name $color $ink $initials -}} {{- return $svg -}}
