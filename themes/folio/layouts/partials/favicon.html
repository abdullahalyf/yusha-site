{{- /* Favicon: SVG with the person's initials in the style accent colour. */}} {{- $p := . -}} {{- $style := site.Params.style | default "harbor" -}} {{- $accent := dict "harbor" "#0E7490" "grove" "#2F6B4F" "orchid" "#B4236B" "graphite" "#1F6FEB" "saffron" "#4338CA" -}} {{- $color := index $accent $style | default "#0E7490" -}} {{- $words := split $p.name " " -}} {{- $initials := "" -}} {{- range $words -}} {{- if $initials | len | lt 2 -}} {{- $initials = printf "%s%s" $initials (substr . 0 1) -}} {{- end -}} {{- end -}} {{- $initials = upper $initials -}} {{- $svg := printf `` $p.name $color $initials -}} {{- return $svg -}}
