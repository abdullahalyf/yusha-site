{{- /* The "Moments" gallery. Images live in personal/photos/gallery/ and are matched by their file name. The order comes from the optional `gallery:` list in profile.yaml, where each entry has a `file` and an optional `caption`: gallery: - file: "ride.jpg" caption: "On the way to the hills" Files that are not in that list are shown after the listed ones, in name order, so a gallery works without any YAML at all. Captions are optional; an image with no caption gets an empty caption. Usage: partial "gallery.html" (dict "profile" $p) */ -}} {{- $p := .profile -}} {{- $files := slice -}} {{- $captions := dict -}} {{- range $p.gallery -}} {{- $name := .file | default "" -}} {{- if $name -}} {{- $files = $files | append $name -}} {{- $captions = merge $captions (dict $name (.caption | default "")) -}} {{- end -}} {{- end -}} {{- /* Every image in the folder, by file name without extension. */ -}} {{- $all := resources.Match "photos/gallery/*" -}} {{- $rest := slice -}} {{- range $all -}} {{- $stem := path.BaseName .Name -}} {{- if not (in $files $stem) -}} {{- $rest = $rest | append $stem -}} {{- end -}} {{- end -}} {{- $rest = sort $rest -}} {{- $files = $files | append $rest -}} {{- if $files -}}
