{{- /* Render hook for markdown images. A post refers to an image by its path inside assets/, for example `blog/my-post-body.jpg`. Hugo resolves that against the mounted assets folders, so the file goes through the same image pipeline as the photos: two WebP widths, explicit dimensions so the browser can reserve the space, and lazy loading below the fold. An image that cannot be found is left as a plain pointing at the original path. That way a typo shows a broken image rather than failing the whole build. */ -}} {{- $src := .Destination -}} {{- $alt := .Text | plainify -}} {{- $resource := "" -}} {{- if not (urls.Parse $src).IsAbs -}} {{- $resource = resources.GetMatch $src -}} {{- end -}} {{- if $resource -}} {{- if and (eq $resource.ResourceType "image") (gt $resource.Width 0) -}} {{- $w1 := 480 -}} {{- $w2 := 960 -}} {{- if lt $resource.Width 480 -}}{{- $w1 = $resource.Width -}}{{- end -}} {{- if lt $resource.Width 960 -}}{{- $w2 = $resource.Width -}}{{- end -}} {{- $src1 := $resource.Resize (printf "%dx webp q82" $w1) -}} {{- $src2 := $resource.Resize (printf "%dx webp q82" $w2) -}} {{- $ratio := div (float $resource.Height) (float $resource.Width) -}} {{- $h := int (math.Round (mul (float $w2) $ratio)) -}} {{- else -}} {{- end -}} {{- else -}} {{- end -}}
